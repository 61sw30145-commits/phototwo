      <!-- Dynamic Frame Cards -->

  <div id="step-3" class="hidden text-center space-y-4 w-full max-w-md">

      <video id="webcam" autoplay playsinline class="w-full h-full object-cover"></video>

      <div id="countdown" class="absolute inset-0 flex items-center justify-center text-6xl font-bold text-white bg-black/30 hidden">3</div>

    </div>

    <button id="btn-capture" onclick="startCaptureSequence()" class="px-6 py-2 bg-stone-800 text-white rounded-full">เริ่มถ่าย (3 ช็อต)</button>

  <!-- Step 5: Print -->

  <div id="step-5" class="hidden text-center space-y-6 w-full max-w-md">

    <h2 class="text-2xl font-light">สั่งปริ้นท์รูปภาพ</h2>

    <div id="print-area" class="flex justify-center">

      <canvas id="final-canvas" class="max-w-full rounded shadow-md"></canvas>

    </div>

    <div class="flex items-center justify-center gap-4">

      <label class="text-sm">จำนวนชุด:</label>

      <input type="number" id="print-copies" value="1" min="1" max="10" class="w-16 p-2 border border-stone-300 rounded text-center">

      <button onclick="triggerPrint()" class="px-6 py-2 bg-stone-800 text-white rounded-full">สั่งปริ้นท์</button>

    </div>

  <script>

    let currentStep = 1;

    let selectedFrame = null;

    let capturedImages = [];

    const frames = [

      { id: 1, name: "มินิมอล ขาว", url: "https://via.placeholder.com/600x1800/ffffff/000000?text=Frame+1" },

      { id: 2, name: "มินิมอล ดำ", url: "https://via.placeholder.com/600x1800/000000/ffffff?text=Frame+2" }

    ];


    function goToStep(step) {

      document.querySelectorAll('[id^="step-"]').forEach(el => el.classList.add('hidden'));

      document.getElementById(`step-${step}`).classList.remove('hidden');

      currentStep = step;


      if (step === 2) renderFrames();

      if (step === 3) startCamera();

    }


    function renderFrames() {

      const container = document.getElementById('frame-list');

      container.innerHTML = frames.map(f => `

        <div onclick="selectFrame('${f.url}')" class="border p-4 rounded-lg cursor-pointer hover:border-stone-800">

          <p class="text-sm">${f.name}</p>

        </div>

      `).join('');

    }


    function selectFrame(url) {

      selectedFrame = url;

      goToStep(3);

    }


    async function startCamera() {

      const stream = await navigator.mediaDevices.getUserMedia({ video: true, audio: false });

      document.getElementById('webcam').srcObject = stream;

    }


    function startCaptureSequence() {

      capturedImages = [];

      captureNext(0);

    }


    function captureNext(index) {

      if (index >= 3) {

        renderFinalCanvas();

        goToStep(5);

        return;

      }

      let count = 3;

      const cdEl = document.getElementById('countdown');

      cdEl.classList.remove('hidden');

      cdEl.innerText = count;


      const timer = setInterval(() => {

        count--;

        if (count > 0) {

          cdEl.innerText = count;

        } else {

          clearInterval(timer);

          cdEl.classList.add('hidden');

          takePhoto();

          captureNext(index + 1);

        }

      }, 1000);

    }


    function takePhoto() {

      const video = document.getElementById('webcam');

      const canvas = document.createElement('canvas');

      canvas.width = video.videoWidth;

      canvas.height = video.videoHeight;

      const ctx = canvas.getContext('2d');

      ctx.drawImage(video, 0, 0);

      capturedImages.push(canvas.toDataURL('image/png'));

    }


    function renderFinalCanvas() {

      const canvas = document.getElementById('final-canvas');

      const ctx = canvas.getContext('2d');

      canvas.width = 600;

      canvas.height = 1800; // อัตราส่วน 1:3 สำหรับ 3 ช็อต


      ctx.fillStyle = "#FFFFFF";

      ctx.fillRect(0, 0, canvas.width, canvas.height);


      // วาดรูป 3 ช็อตลงบน Canvas

      capturedImages.forEach((imgSrc, i) => {

        const img = new Image();

        img.onload = () => {

          ctx.drawImage(img, 50, 50 + (i * 550), 500, 500);

        };

        img.src = imgSrc;

      });

    }


    function triggerPrint() {

      const copies = document.getElementById('print-copies').value;

      for (let i = 0; i < copies; i++) {

        window.print();

      }

    }

  </script>
